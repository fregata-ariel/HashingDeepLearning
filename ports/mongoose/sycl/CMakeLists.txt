cmake_minimum_required(VERSION 3.20)
project(mongoose_sycl LANGUAGES CXX)
find_package(AdaptiveCpp CONFIG REQUIRED)
option(HASHING_SYCL_REQUIRE_NVIDIA "Require NVIDIA GPU; never fall back to CPU" OFF)
if(HASHING_SYCL_REQUIRE_NVIDIA)
  add_compile_definitions(HASHING_SYCL_REQUIRE_NVIDIA=1)
endif()
if(NOT TARGET hashing_sycl::common)
  add_subdirectory(../../sycl-common sycl-common)
endif()
find_package(OpenMP REQUIRED)
add_executable(mongoose_sycl_fixture tests/packing.cpp ../lsh_lib/LSH.cpp)
target_compile_features(mongoose_sycl_fixture PRIVATE cxx_std_17)
target_include_directories(mongoose_sycl_fixture PRIVATE include ../lsh_lib)
target_link_libraries(mongoose_sycl_fixture PRIVATE hashing_sycl::common OpenMP::OpenMP_CXX)
add_sycl_to_target(TARGET mongoose_sycl_fixture SOURCES tests/packing.cpp)
enable_testing()
if(HASHING_SYCL_REQUIRE_NVIDIA)
  add_test(NAME mongoose_sycl_nvidia COMMAND mongoose_sycl_fixture)
else()
  add_test(NAME mongoose_sycl_cpu COMMAND mongoose_sycl_fixture)
endif()
