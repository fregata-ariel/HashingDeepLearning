cmake_minimum_required(VERSION 3.20)
project(scatterbrain_sycl LANGUAGES CXX)
find_package(AdaptiveCpp CONFIG REQUIRED)
if(NOT TARGET hashing_sycl::common)
 add_subdirectory(../../sycl-common sycl-common)
endif()
add_library(scatterbrain_sycl SHARED adapter.cpp)
target_compile_features(scatterbrain_sycl PUBLIC cxx_std_17)
target_include_directories(scatterbrain_sycl PUBLIC include)
target_link_libraries(scatterbrain_sycl PRIVATE hashing_sycl::common)
add_sycl_to_target(TARGET scatterbrain_sycl SOURCES adapter.cpp)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()
add_test(NAME scatterbrain_sycl_cpu COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/validate.py --library $<TARGET_FILE:scatterbrain_sycl> --evidence ${CMAKE_CURRENT_BINARY_DIR}/evidence.json)
