cmake_minimum_required(VERSION 3.20)
project(scatterbrain_sycl LANGUAGES CXX)
find_package(AdaptiveCpp CONFIG REQUIRED)
option(HASHING_SYCL_REQUIRE_NVIDIA "Require NVIDIA GPU; never fall back to CPU" OFF)
if(HASHING_SYCL_REQUIRE_NVIDIA)
  add_compile_definitions(HASHING_SYCL_REQUIRE_NVIDIA=1)
endif()
if(NOT TARGET hashing_sycl::common)
 add_subdirectory(../../sycl-common sycl-common)
endif()
add_library(scatterbrain_sycl SHARED adapter.cpp)
target_compile_features(scatterbrain_sycl PUBLIC cxx_std_17)
target_include_directories(scatterbrain_sycl PUBLIC include)
target_link_libraries(scatterbrain_sycl PRIVATE hashing_sycl::common)
add_sycl_to_target(TARGET scatterbrain_sycl SOURCES adapter.cpp)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
enable_testing()
set(backend_args)
if(HASHING_SYCL_REQUIRE_NVIDIA)
  list(APPEND backend_args --gpu-backend)
  set(validation_name scatterbrain_sycl_nvidia)
else()
  set(validation_name scatterbrain_sycl_cpu)
endif()
add_test(NAME ${validation_name} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/validate.py --library $<TARGET_FILE:scatterbrain_sycl> --evidence ${CMAKE_CURRENT_BINARY_DIR}/evidence.json ${backend_args})
add_test(NAME scatterbrain_sycl_threads COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/threads.py --library $<TARGET_FILE:scatterbrain_sycl>)

# Keep the foreign-thread cache independent so first invocation exercises cold JIT.
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/threads-appdb")
set_tests_properties(scatterbrain_sycl_threads PROPERTIES ENVIRONMENT "ACPP_APPDB_DIR=${CMAKE_CURRENT_BINARY_DIR}/threads-appdb")
