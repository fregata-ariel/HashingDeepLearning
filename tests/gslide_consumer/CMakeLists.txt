cmake_minimum_required(VERSION 3.20)
project(gslide_external_consumer LANGUAGES CXX)
option(GSLIDE_CONSUMER_SYCL "Build the real SYCL consumer as well as the SDK-free consumer" OFF)
option(GSLIDE_CONSUMER_REQUIRE_NVIDIA "Reject CPU fallback" OFF)
find_package(GSlide 0.1.0 EXACT CONFIG REQUIRED COMPONENTS control)
enable_testing()
add_executable(consumer_host host.cpp host_other.cpp)
target_link_libraries(consumer_host PRIVATE gslide::control)
add_test(NAME consumer.host COMMAND consumer_host)
if(GSLIDE_CONSUMER_SYCL)
  find_package(GSlide 0.1.0 EXACT CONFIG REQUIRED COMPONENTS sycl)
  find_package(AdaptiveCpp CONFIG REQUIRED)
  add_executable(consumer_sycl sycl.cpp sycl_other.cpp)
  target_link_libraries(consumer_sycl PRIVATE gslide::sycl)
  if(GSLIDE_CONSUMER_REQUIRE_NVIDIA)
    target_compile_definitions(consumer_sycl PRIVATE GSLIDE_CONSUMER_REQUIRE_NVIDIA=1)
  endif()
  add_sycl_to_target(TARGET consumer_sycl SOURCES sycl.cpp sycl_other.cpp)
  add_test(NAME consumer.sycl COMMAND consumer_sycl)
endif()
