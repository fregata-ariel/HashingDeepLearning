cmake_minimum_required(VERSION 3.20)
project(gslide_host_contracts LANGUAGES CXX)
if(NOT TARGET gslide::control)
  add_subdirectory(../../ports/g-slide/library gslide-library)
endif()
enable_testing()
foreach(suite control batch)
  add_executable(gslide_host_${suite} test_${suite}.cpp)
  target_link_libraries(gslide_host_${suite} PRIVATE gslide::control)
  add_test(NAME gslide.host.${suite} COMMAND gslide_host_${suite})
endforeach()
