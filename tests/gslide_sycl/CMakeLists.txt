cmake_minimum_required(VERSION 3.20)
project(gslide_sycl LANGUAGES CXX)
find_package(AdaptiveCpp CONFIG REQUIRED)
add_subdirectory(../../ports/g-slide/library gslide-library)
option(HASHING_SYCL_REQUIRE_NVIDIA "Require NVIDIA GPU; never fall back to CPU" OFF)
if(HASHING_SYCL_REQUIRE_NVIDIA)
  add_compile_definitions(HASHING_SYCL_REQUIRE_NVIDIA=1)
endif()
enable_testing()
add_subdirectory(../gslide_host host)
function(gslide_device_target target source)
  add_executable(${target} ${source})
  target_link_libraries(${target} PRIVATE gslide::sycl)
  target_include_directories(${target} PRIVATE ../../ports/sycl-common/include)
  add_sycl_to_target(TARGET ${target} SOURCES ${source})
endfunction()
# Legacy aggregate remains buildable; CTest runs each component once instead.
gslide_device_target(gslide_sycl_test test_kernels.cpp)
foreach(suite reductions dense_ids random hash candidates operators memory optimizer output selection)
  gslide_device_target(gslide_${suite}_test test_${suite}.cpp)
  add_test(NAME gslide.sycl.${suite} COMMAND gslide_${suite}_test)
endforeach()
gslide_device_target(gslide_sycl_training_test test_network.cpp)
add_test(NAME gslide.sycl.network COMMAND gslide_sycl_training_test)
